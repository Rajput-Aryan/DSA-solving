class Solution {
public:
    long long countCommas(long long n) {

        if (n < 1000) {
            return 0;
        } 

        int ans=0;
        int digit = 0;
        int dup = n;
        

        while (n > 10) {
            n /= 10; // n =n/10;
            digit++;
        }

        int index=digit/3;

        while(index>0){
            int fu=pow(10, 3*index)-1;
            ans = ans + (dup - fu)*index;
            dup = fu;
            index--;
        }

        return ans;
        
    }
};
